# Inlay Component Model Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a componentized inlay system where finishing a jade body into a ring, pendant, earring, cord piece, bracelet, or handpiece produces semantic finished jewelry and renders visible mounts, cords, metals, and accent stones.

**Architecture:** The server owns authoritative finished-product semantics by converting `finishPlan` and `quote` into a `components` array inside `applyInlay()`. The browser treats products with `components` as layered jewelry, using a shared SVG renderer from `productCanvas()` and the finish-plan modal preview. Order and display helpers read `productType || kind` so finished jewelry is gameplay-visible as the final jewelry type.

**Tech Stack:** Node.js ES modules, existing smoke-test scripts with `node:assert/strict`, browser-side vanilla JavaScript, SVG markup generated by a shared browser module.

## Global Constraints

- No old-data compatibility: products produced before this model may be treated as invalid debug data.
- No new external bitmap, 3D, or asset pipeline dependency.
- Keep existing finish modal fields: product type, setting, metal, polish, gem, gem count, backing.
- The finished product's `kind` and `productType` must become the final jewelry type.
- The jade body's original form must live inside a `components` entry, not remain the finished product semantic type.
- Cord inlay uses a cord component and no metal mount.
- Non-cord inlay uses a mount component; accent stones are present only when `gemCount > 0`.
- Use TDD: every production behavior change starts with a failing smoke assertion.

function productTemplateSize(product) {
  const size = product?.templateSize;
  if (!size || typeof size !== "object") return null;
  const w = Number(size.w);
  const h = Number(size.h);
  return Number.isFinite(w) && Number.isFinite(h) ? { w, h } : null;
}

export function buildInlayComponents(product, plan, quote = {}) {
  const components = [{
    type: "jade",
    role: "mainStone",
    kind: itemShape(product) || product?.kind || product?.design || null,
    sourceMaterial: product?.sourceMaterial ? structuredClone(product.sourceMaterial) : null,
    templateSize: productTemplateSize(product)
  }];

  if (plan.settingId === "cord") {
    components.push({
      type: "cord",
      role: "hanger",
      productType: plan.productType,
      settingId: plan.settingId,
      polishId: plan.polishId
    });
  } else {
    components.push({
      type: "mount",
      role: "setting",
      productType: plan.productType,
      settingId: plan.settingId,
      metalId: plan.metalId,
      backingId: plan.backingId || null,
      metalAmount: quote.metalAmount || 0
    });
  }

  if (Number(plan.gemCount || 0) > 0) {
    components.push({
      type: "accentStones",
      role: plan.settingId === "channel" ? "channel" : "halo",
      gemId: plan.gemId,
      count: Number(plan.gemCount || 0)
    });
  }

  return components;
}

function escapeAttr(value) {
  return String(value ?? "")
    .replace(/&/g, "&amp;")
    .replace(/"/g, "&quot;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;");
}

const JEWELRY_METALS = {
  goldPure: { a: "#f3c650", b: "#fff2ad", c: "#a87516" },
  gold18k: { a: "#d99a28", b: "#ffe08a", c: "#8f5d14" },
  silver925: { a: "#c8d1d5", b: "#ffffff", c: "#7d8b92" },
  platinum: { a: "#b8c4ca", b: "#f7fbff", c: "#6f7d85" }
};

const JEWELRY_GEMS = {
  diamond: "#f8fbff",
  ruby: "#c8234a",
  sapphire: "#2459c8"
};

function jewelryComponent(item, type) {
  return Array.isArray(item?.components)
    ? item.components.find((component) => component?.type === type)
    : null;
}

function componentJadeMarkup(item) {
  const jade = jewelryComponent(item, "jade") || {};
  return [
    '<g class="jewelry-jade" data-jade-component-kind="' + escapeAttr(jade.kind || "cabochon") + '">',
    '<ellipse cx="80" cy="68" rx="24" ry="30" fill="url(#jadeGradient)" stroke="rgba(14,84,64,.72)" stroke-width="2.6"/>',
    '<path d="M66 54C75 42 91 46 98 60" fill="none" stroke="rgba(255,255,255,.42)" stroke-width="3" stroke-linecap="round"/>',
    '<path d="M61 72C74 82 89 82 101 70" fill="none" stroke="rgba(20,111,81,.25)" stroke-width="5" stroke-linecap="round"/>',
    "</g>"
  ].join("");
}

function componentAccentMarkup(item) {
  const accent = jewelryComponent(item, "accentStones");
  const count = Math.min(12, Math.max(0, Number(accent?.count || 0)));
  if (!accent || count <= 0) return "";
  const fill = JEWELRY_GEMS[accent.gemId] || "#f8fbff";
  const stones = [];
  for (let index = 0; index < count; index += 1) {
    const angle = -Math.PI * 0.82 + (index / Math.max(1, count - 1)) * Math.PI * 1.64;
    const x = 80 + Math.cos(angle) * 36;
    const y = 68 + Math.sin(angle) * 34;
    stones.push('<circle class="jewelry-accent-stone" cx="' + x.toFixed(1) + '" cy="' + y.toFixed(1) + '" r="3.4" fill="' + fill + '" stroke="rgba(255,255,255,.86)" stroke-width="1"/>');
  }
  return '<g class="jewelry-accent-stones" data-jewelry-gem="' + escapeAttr(accent.gemId || "") + '">' + stones.join("") + "</g>";
}

function componentMountMarkup(item, ids) {
  const mount = jewelryComponent(item, "mount");
  if (!mount) return "";
  const metal = JEWELRY_METALS[mount.metalId] || JEWELRY_METALS.gold18k;
  const productType = mount.productType || item.productType || item.kind || "ring";
  const stroke = 'stroke="url(#' + ids.metal + ')" stroke-linecap="round" stroke-linejoin="round"';
  const pieces = ['<g class="jewelry-metal-mount" data-jewelry-metal="' + escapeAttr(mount.metalId || "") + '" data-jewelry-setting="' + escapeAttr(mount.settingId || "") + '">'];
  if (productType === "ring") {
    pieces.push('<ellipse class="jewelry-ring-band" cx="80" cy="102" rx="42" ry="24" fill="none" ' + stroke + ' stroke-width="12"/>');
    pieces.push('<ellipse cx="80" cy="102" rx="30" ry="15" fill="rgba(255,255,255,.45)" opacity=".34"/>');
  } else if (productType === "earring") {
    pieces.push('<path class="jewelry-ear-hook" d="M79 36C99 18 121 38 98 55" fill="none" ' + stroke + ' stroke-width="8"/>');
  } else {
    pieces.push('<path class="jewelry-bail" d="M67 31C67 16 93 16 93 31L88 49L72 49Z" fill="url(#' + ids.metal + ')" stroke="' + metal.c + '" stroke-width="2"/>');
  }
  if (mount.backingId === "closed") {
    pieces.push('<ellipse class="jewelry-closed-backing" cx="80" cy="70" rx="28" ry="33" fill="url(#' + ids.metal + ')" opacity=".72"/>');
  }
  if (mount.settingId === "bezel" || mount.settingId === "channel" || mount.settingId === "invisible") {
    pieces.push('<ellipse class="jewelry-bezel" cx="80" cy="68" rx="30" ry="36" fill="none" ' + stroke + ' stroke-width="7"/>');
  } else {
    for (const [x, y, rot] of [[56, 42, -34], [104, 42, 34], [55, 94, 35], [105, 94, -35]]) {
      pieces.push('<path class="jewelry-prong" d="M' + x + ' ' + y + 'l' + (x < 80 ? 10 : -10) + ' ' + (y < 68 ? 13 : -13) + '" fill="none" ' + stroke + ' stroke-width="5" transform="rotate(' + rot + ' ' + x + ' ' + y + ')"/>');
    }
  }
  pieces.push("</g>");
  return pieces.join("");
}

function componentCordMarkup(item) {
  const cord = jewelryComponent(item, "cord");
  if (!cord) return "";
  return [
    '<g class="jewelry-cord" data-jewelry-setting="cord">',
    '<path d="M35 12C58 28 102 28 125 12" fill="none" stroke="#2d4938" stroke-width="6" stroke-linecap="round"/>',
    '<path d="M63 30C55 46 55 56 66 66M97 30C105 46 105 56 94 66" fill="none" stroke="#17291f" stroke-width="5" stroke-linecap="round"/>',
    '<circle cx="72" cy="42" r="5" fill="#526f46"/><circle cx="88" cy="42" r="5" fill="#526f46"/>',
    "</g>"
  ].join("");
}

export function componentJewelrySvg(item, size = "finished") {
  const mount = jewelryComponent(item, "mount");
  const ids = {
    metal: `metal-${escapeAttr(item?.id || "product")}`,
    jade: `jade-${escapeAttr(item?.id || "product")}`
  };
  const metal = JEWELRY_METALS[mount?.metalId] || JEWELRY_METALS.gold18k;
  const productType = item?.productType || item?.kind || mount?.productType || "product";
  return `
    <div class="component-jewelry-product ${escapeAttr(size)}" role="img" aria-label="${escapeAttr(item?.name || "成品")}" data-jewelry-product-type="${escapeAttr(productType)}" data-jewelry-metal="${escapeAttr(mount?.metalId || "")}" data-jewelry-setting="${escapeAttr(mount?.settingId || jewelryComponent(item, "cord")?.settingId || "")}" data-jewelry-gem="${escapeAttr(jewelryComponent(item, "accentStones")?.gemId || "")}">
      <svg class="component-jewelry-svg" viewBox="0 0 160 150" preserveAspectRatio="xMidYMid meet" aria-hidden="true" focusable="false">
        <defs>
          <linearGradient id="${ids.metal}" x1="0" y1="0" x2="1" y2="1"><stop offset="0" stop-color="${metal.b}"/><stop offset=".45" stop-color="${metal.a}"/><stop offset="1" stop-color="${metal.c}"/></linearGradient>
          <radialGradient id="${ids.jade}" cx=".38" cy=".28" r=".78"><stop offset="0" stop-color="#d8fff1"/><stop offset=".48" stop-color="#3eb27c"/><stop offset="1" stop-color="#0f6d52"/></radialGradient>
          <radialGradient id="jadeGradient" cx=".38" cy=".28" r=".78"><stop offset="0" stop-color="#d8fff1"/><stop offset=".48" stop-color="#3eb27c"/><stop offset="1" stop-color="#0f6d52"/></radialGradient>
        </defs>
        ${componentCordMarkup(item)}
        ${componentMountMarkup(item, ids)}
        ${componentJadeMarkup(item)}
        ${componentAccentMarkup(item)}
      </svg>
    </div>
  `;
}
